cmake_minimum_required(VERSION 3.20)
project(drs_tests C)
enable_testing()
add_executable(drs_tests test_drs.c ../../src/core/drs.c)
target_include_directories(drs_tests PRIVATE ../../src)
set_target_properties(drs_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_options(drs_tests PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME drs_frames COMMAND drs_tests)

foreach(header IN ITEMS zephyr/drivers/uart.h zephyr/kernel.h zephyr/logging/log.h zephyr/sys/atomic.h zephyr/sys/util.h)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stubs/${header}" CONTENT "#include \"mock.h\"\n")
endforeach()
add_executable(drs_io_tests test_io.c ../../src/core/drs.c)
target_include_directories(drs_io_tests PRIVATE . ../../src "${CMAKE_CURRENT_BINARY_DIR}/stubs")
set_target_properties(drs_io_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_options(drs_io_tests PRIVATE -Wall -Wextra -Werror -pedantic)
foreach(case IN ITEMS receive corrupt exception timeout tx_failure rx_error reconnect init_error)
  add_test(NAME drs_io_${case} COMMAND drs_io_tests ${case})
endforeach()
