cmake_minimum_required(VERSION 3.20)
project(output_tests C)
enable_testing()
foreach(header IN ITEMS zephyr/drivers/gpio.h zephyr/drivers/watchdog.h zephyr/kernel.h zephyr/sys/util.h)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stubs/${header}" CONTENT "#include \"mock.h\"\n")
endforeach()
add_executable(output_tests test_output.c ../../src/core/power.c)
target_include_directories(output_tests PRIVATE . ../../src "${CMAKE_CURRENT_BINARY_DIR}/stubs")
target_compile_definitions(output_tests PRIVATE CONFIG_HT_OUTPUT=1)
set_target_properties(output_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(output_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
foreach(case IN ITEMS stop_priority interlocks stale_ui sample_fault standby_sampling standby_board_fault init_failure
    sample_permission pending_sample_invalid first_window_grace
    protection_race stop_in_flight relay_fault shutdown_output)
  add_test(NAME output_${case} COMMAND output_tests ${case})
endforeach()
