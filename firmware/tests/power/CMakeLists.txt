cmake_minimum_required(VERSION 3.20)
project(power_tests C)
enable_testing()
add_executable(power_tests test_power.c ../../src/core/power.c ../../src/core/rms.c ../../src/drivers/tca9539_safe.c)
target_include_directories(power_tests PRIVATE ../../src)
set_target_properties(power_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(power_tests PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(power_tests PRIVATE m)
endif()
foreach(case IN ITEMS sequence timing all_ranges stop_stages interlock_stages io_failure stale
    regulation limits open_short continuous rms_scaling rms_clipping relay_mutex)
  add_test(NAME power_${case} COMMAND power_tests ${case})
endforeach()
